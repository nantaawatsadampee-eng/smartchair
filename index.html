function onResults(results) { if (!results.poseLandmarks) return; const landmarks = results.poseLandmarks; // ดึงพิกัดจุดที่จำเป็น const leftEar = landmarks[7]; const rightEar = landmarks[8]; const leftShoulder = landmarks[11]; const rightShoulder = landmarks[12]; // ------------------------------------------------------------- // 1. ตรวจจับการนั่งเอียง (Shoulder Tilt) // ------------------------------------------------------------- const shoulderDiff = Math.abs(leftShoulder.y - rightShoulder.y); const isTilt = shoulderDiff > 0.045; // ปรับค่าได้ (0.03 - 0.06) ยิ่งน้อยยิ่งไว // ------------------------------------------------------------- // 2. ตรวจจับหลังค่อม / ก้มคอ (Slouching) // ------------------------------------------------------------- const avgEarY = (leftEar.y + rightEar.y) / 2; const avgShoulderY = (leftShoulder.y + rightShoulder.y) / 2; const neckDistance = avgShoulderY - avgEarY; // ระยะห่างจากหูถึงไหล่ const isSlouch = neckDistance < 0.12; // หากระยะห่างน้อยกว่า 0.12 แสดงว่าหลังค่อม/ก้มคอ // ------------------------------------------------------------- // สรุปสถานะและส่งข้อมูล BLE ไปยัง ESP32 // ------------------------------------------------------------- const isBadPosture = isTilt || isSlouch; const statusElement = document.getElementById("statusText"); // ปรับ ID ให้ตรงกับหน้าเว็บของคุณ if (isBadPosture) { let msg = "ท่านั่ง: "; if (isTilt && isSlouch) msg += "เอียง + หลังค่อม ⚠️"; else if (isTilt) msg += "นั่งเอียง ⚠️"; else if (isSlouch) msg += "หลังค่อม / ก้มคอ ⚠️"; statusElement.innerText = msg; statusElement.style.color = "red"; sendBLEData("1"); // ส่ง '1' ไปแจ้ง ESP32 ว่านั่งผิดท่า } else { statusElement.innerText = "ท่านั่ง: ปกติ ✅"; statusElement.style.color = "green"; sendBLEData("0"); // ส่ง '0' ไปแจ้ง ESP32 ว่านั่งปกติ } }
