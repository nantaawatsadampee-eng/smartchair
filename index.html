let isFsrError = false; // ตัวแปรเก็บสถานะเซนเซอร์ FSR // ฟังก์ชันเมื่อเชื่อมต่อ BLE สำเร็จ ให้เปิดรับการแจ้งเตือนจาก ESP32 async function connectBLE() { try { const device = await navigator.bluetooth.requestDevice({ filters: [{ name: 'SmartChair_ESP32' }], optionalServices: ['4fafc201-1fb5-459e-8fcc-c5c9c331914b'] }); const server = await device.gatt.connect(); const service = await server.getPrimaryService('4fafc201-1fb5-459e-8fcc-c5c9c331914b'); bleCharacteristic = await service.getCharacteristic('beb5483e-36e1-4688-b7f5-ea07361b26a8'); // เริ่มรับ Notification จาก ESP32 await bleCharacteristic.startNotifications(); bleCharacteristic.addEventListener('characteristicvaluechanged', (event) => { const value = new TextDecoder().decode(event.target.value); if (value === 'E') { isFsrError = true; // เซนเซอร์ไม่ถูกสัมผัสครบเกิน 5 วินาที } else { isFsrError = false; // เซนเซอร์สัมผัสปกติ } updateStatusUI(); }); document.getElementById("bleStatus").innerText = "สถานะ BLE: เชื่อมต่อสำเร็จเรียบร้อย! ✅"; } catch (error) { console.error(error); } } // ฟังก์ชันอัปเดตข้อความบนหน้าจอเว็บ function updateStatusUI(isAiBad = false, aiMsg = "") { const statusElement = document.getElementById("statusText"); // หาก FSR เตือน (สัมผัสไม่ครบเกิน 5 วิ) จะขึ้นเตือนเรื่องเซนเซอร์เป็นลำดับแรก if (isFsrError) { statusElement.innerText = "ท่านั่ง: เซนเซอร์ถูกสัมผัสไม่ครบทุกจุด ⚠️"; statusElement.style.color = "red"; } else if (isAiBad) { statusElement.innerText = "ท่านั่ง: " + aiMsg; statusElement.style.color = "red"; } else { statusElement.innerText = "ท่านั่ง: ปกติ ✅"; statusElement.style.color = "green"; } }
